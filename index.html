// SCRIPT INI SEKARANG BERTINDAK SEBAGAI API UNTUK GITHUB // Menerima permintaan POST dari GitHub (Frontend) function doPost(e) { try { var requestData = JSON.parse(e.postData.contents); var action = requestData.action; var result = {}; if (action === "login") { result = verifyLogin(requestData.username, requestData.password); } else if (action === "saveTemplate") { result = saveTemplateToServer(requestData.username, requestData.template); } else { result = { success: false, message: "Aksi tidak dikenali" }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Error Server: " + error.message })) .setMimeType(ContentService.MimeType.JSON); } } // ========================================== // LOGIKA ASLI DI BAWAH INI TIDAK BERUBAH // ========================================== function verifyLogin(username, password) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetUsers = ss.getSheetByName("Users"); var sheetData = ss.getSheetByName("Data"); if (!sheetUsers) return { success: false, message: "Sheet 'Users' tidak ditemukan!" }; var usersData = sheetUsers.getDataRange().getValues(); var globalRenewLink = "#"; var globalDownloadLink = "#"; if (sheetData != null) { var configData = sheetData.getDataRange().getValues(); if (configData.length > 1) { globalRenewLink = String(configData[1][0] || "#").trim(); globalDownloadLink = String(configData[1][1] || "#").trim(); } } for (var i = 1; i < usersData.length; i++) { var sheetUsername = String(usersData[i][0] || "").trim(); var sheetPassword = String(usersData[i][1] || "").trim(); var inputUsername = String(username || "").trim(); var inputPassword = String(password || "").trim(); if (sheetUsername === "" && sheetPassword === "") continue; if (sheetUsername === inputUsername && sheetPassword === inputPassword) { var rawExpiry = usersData[i][4]; var safeExpiryDate = rawExpiry ? String(rawExpiry) : ""; return { success: true, template: String(usersData[i][2] || ""), licenseCode: String(usersData[i][3] || "-"), expiryDate: safeExpiryDate, renewLink: globalRenewLink, downloadLink: globalDownloadLink }; } } return { success: false, message: "Username atau Password salah!" }; } function saveTemplateToServer(username, newTemplate) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Users"); if (!sheet) return { success: false, message: "Sheet error" }; var data = sheet.getDataRange().getValues(); var inputUsername = String(username).trim(); for (var i = 1; i < data.length; i++) { var sheetUsername = String(data[i][0]).trim(); if (sheetUsername === inputUsername) { sheet.getRange(i + 1, 3).setValue(newTemplate); return { success: true }; } } return { success: false, message: "User tidak ditemukan" }; }
